1. Who are the employees that are working in the Marketing department who make more
than 100K and can be called as baby boomers?
SELECT
    e.first_name,
    e.last_name,
    e.gender,
    TO_CHAR(e.hire_date, 'YYYY-MM-DD')     AS hire_date,
    TO_CHAR(e.birth_date, 'YYYY-MM-DD')    AS birth_date,
    s.amount                               AS salary,
    d.dept_name                            AS department,
    e.id                                   AS employee_id
FROM
    seerworks_org.employee           AS e
JOIN
    seerworks_org.salary             AS s  ON e.id = s.employee_id
JOIN
    seerworks_org.department_employee AS de ON e.id = de.employee_id
JOIN
    seerworks_org.department         AS d  ON de.department_id = d.id
WHERE
    d.dept_name = 'Marketing'
    AND e.birth_date BETWEEN '1946-01-01' AND '1964-12-31'
    AND s.amount > 100000
    AND s.to_date  > CURRENT_DATE
    AND de.to_date > CURRENT_DATE
ORDER BY
    e.last_name, e.first_name;   -- optional: alphabetical sort

9. If 5 new employees were to be hired in Sales Department with each getting a pay of $50000, then how much
would the average pay of all the employees in Sales department change?
WITH current_sales_data AS (
    SELECT
        SUM(s.amount) AS total_current_salary,
        COUNT(*) AS current_employee_count
    FROM seerworks_org.salary s
    JOIN seerworks_org.department_employee de ON s.employee_id = de.employee_id
    JOIN seerworks_org.department d ON de.department_id = d.id
    WHERE d.dept_name = 'Sales'
    AND de.to_date > CURRENT_DATE
    AND s.to_date > CURRENT_DATE
)
SELECT
    ( (total_current_salary + 250000) / (current_employee_count + 5) ) -
    (total_current_salary / current_employee_count) AS average_pay_change
FROM current_sales_data

10. Can you find the difference in expenses when it comes to salaries that
are paid to employees that work in Marketing and Sales department?

WITH dept_salaries AS (
    SELECT
        d.dept_name,
        SUM(s.amount) AS total_salary
    FROM seerworks_org.salary s
    JOIN seerworks_org.department_employee de ON s.employee_id = de.employee_id
    JOIN seerworks_org.department d ON de.department_id = d.id
    WHERE d.dept_name IN ('Marketing', 'Sales')
      AND de.to_date > CURRENT_DATE
      AND s.to_date > CURRENT_DATE
    GROUP BY d.dept_name
)
SELECT
    COALESCE(MAX(CASE WHEN dept_name = 'Sales'     THEN total_salary END), 0) -
    COALESCE(MAX(CASE WHEN dept_name = 'Marketing' THEN total_salary END), 0) AS salary_difference
FROM dept_salaries;

11. Who are the top 10 employees in Sales department that are close to retirement? The retirement age is 65
SELECT
    e.first_name,
    e.last_name,
    e.gender,
    TO_CHAR(e.hire_date, 'YYYY-MM-DD')     AS hire_date,
    TO_CHAR(e.birth_date, 'YYYY-MM-DD')    AS birth_date,
    EXTRACT(YEAR FROM AGE(CURRENT_DATE, e.birth_date)) AS approximate_age,
    e.id                                   AS employee_id
FROM seerworks_org.employee           AS e
JOIN seerworks_org.department_employee AS de ON e.id = de.employee_id
JOIN seerworks_org.department         AS d  ON de.department_id = d.id
WHERE d.dept_name = 'Sales'
  AND de.to_date > CURRENT_DATE
  AND EXTRACT(YEAR FROM AGE(CURRENT_DATE, e.birth_date)) >= 60   -- or BETWEEN 60 AND 64
ORDER BY e.birth_date DESC   -- oldest first = closest to retirement
LIMIT 10;

12. What is the average salary for employees per department based on their titles
SELECT
    d.dept_name,
    t.title,
    AVG(s.amount) AS average_salary
FROM seerworks_org.salary s
JOIN seerworks_org.employee e ON s.employee_id = e.id
JOIN seerworks_org.department_employee de ON e.id = de.employee_id
JOIN seerworks_org.department d ON de.department_id = d.id
JOIN seerworks_org.title t ON e.id = t.employee_id
WHERE s.to_date > CURRENT_DATE
AND de.to_date > CURRENT_DATE
AND t.to_date > CURRENT_DATE
GROUP BY d.dept_name, t.title
ORDER BY d.dept_name, t.title;

                    ) AS rn

                JOIN seerworks_org.department d

            )
            SELECT
                SUM(amount) AS total_savings
            FROM ranked_salaries
            WHERE rn <= 10;

18. If the top 20 earning employees from each department leave, how much can the company save over the next 5 years?
WITH ranked_salaries AS (
    SELECT
        d.dept_name,
        s.amount,
        ROW_NUMBER() OVER (
            PARTITION BY d.dept_name
            ORDER BY s.amount DESC
        ) AS rn
    FROM seerworks_org.salary s
    JOIN seerworks_org.department_employee de ON s.employee_id = de.employee_id
    JOIN seerworks_org.department d ON de.department_id = d.id
    WHERE de.to_date > CURRENT_DATE
      AND s.to_date > CURRENT_DATE
)
SELECT
    ROUND(SUM(amount) * 5, 2) AS estimated_five_year_savings
FROM ranked_salaries
WHERE rn <= 20;
