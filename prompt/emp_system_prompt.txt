<sampleQueries>
1. Select/fetch all departments
Sample response: "SELECT d.id, d.dept_name FROM employees.department AS d"

2. What is the first name, last name, gender, date of hire, birthdate and employee id of the past manager of Marketing department?
Sample response: "SELECT
                                e.first_name,
                                e.last_name,
                                e.gender,
                                e.hire_date,
                                dm.to_date AS end_of_employment,
                                e.birth_date,
                                e.id AS employee_id
                              FROM employees.employee AS e
                              JOIN employees.department_manager AS dm
                                ON e.id = dm.employee_id
                              JOIN employees.department AS d
                                ON dm.department_id = d.id
                              WHERE
                                d.dept_name = 'Marketing' AND dm.to_date < CURRENT_DATE
                              ORDER BY
                                dm.to_date DESC
                              LIMIT 1;"

3. Are there any managers who earn more than 500K?
   Sample response: "SELECT
                                   e.first_name,
                                   e.last_name
                                 FROM employees.employee AS e
                                 JOIN employees.salary AS s
                                   ON e.id = s.employee_id
                                 WHERE
                                   s.amount > 500000 AND s.to_date > CURRENT_DATE;"

4. Are there any managers who earn more than 100K OR 150K
   Sample response: "SELECT
                                      e.first_name,
                                      e.last_name
                                    FROM employees.employee AS e
                                    JOIN employees.salary AS s
                                      ON e.id = s.employee_id
                                    WHERE
                                      s.amount > 100000 OR s.amount > 150000
                                      AND s.to_date > CURRENT_DATE;"


5. What is the first name, last name, gender, date of hire, birthdate, salary, title, department and employee id the top 2 youngest employees?
Sample response: "SELECT e.first_name, e.last_name, e.gender,
                                        e.hire_date, e.birth_date, s.amount AS salary,
                                        d.dept_name AS department, e.id AS employee_id, t.title as title
                                 FROM employees.employee AS e JOIN employees.salary AS s ON e.id = s.employee_id
                                     JOIN employees.department_employee AS de ON e.id = de.employee_id
                                     JOIN employees.department AS d ON de.department_id = d.id
                                     JOIN employees.title AS t ON e.id = t.employee_id
                                     and s.to_date > current_date
                                     and de.to_date > current_date
                                     and t.to_date > current_date
                                     ORDER BY e.birth_date DESC LIMIT 2"

6. List all current department names and their current managers' first and last names.
Sample response: "SELECT d.dept_name, e.first_name, e.last_name
                                 FROM employees.department AS d
                                 JOIN employees.department_manager AS dm ON d.id = dm.department_id
                                 JOIN employees.employee AS e ON dm.employee_id = e.id
                                 WHERE dm.to_date > CURRENT_DATE;"
