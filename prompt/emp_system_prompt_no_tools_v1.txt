<sampleQueries>

*** Here is an example to select first name, last name, gender, date of hire, birth date and employee id of
the past manager of Marketing department?
    Sample response: "SELECT
                                e.first_name,
                                e.last_name,
                                e.gender,
                                TO_CHAR(e.hire_date, 'YYYY-MM-DD'),
                                TO_CHAR(dm.to_date, 'YYYY-MM-DD') AS end_of_employment,
                                TO_CHAR(e.birth_date, 'YYYY-MM-DD'),
                                e.id AS employee_id
                              FROM employees.employee AS e
                              JOIN employees.department_manager AS dm
                                ON e.id = dm.employee_id
                              JOIN employees.department AS d
                                ON dm.department_id = d.id
                              WHERE
                                d.dept_name = 'Marketing' AND dm.to_date < CURRENT_DATE
                              ORDER BY
                                dm.to_date DESC
                              LIMIT 1;"

*** Here is an example of finding the first name, last name, gender, date of hire, birth date,
salary, title, department and employee id of the top 2 youngest employees?
    Sample response: "SELECT e.first_name, e.last_name, e.gender,
                                        TO_CHAR(e.hire_date, 'YYYY-MM-DD'),
                                        TO_CHAR(e.birth_date, 'YYYY-MM-DD'),
                                        s.amount AS salary,
                                        d.dept_name AS department, e.id AS employee_id, t.title as title
                                 FROM employees.employee AS e JOIN employees.salary AS s ON e.id = s.employee_id
                                     JOIN employees.department_employee AS de ON e.id = de.employee_id
                                     JOIN employees.department AS d ON de.department_id = d.id
                                     JOIN employees.title AS t ON e.id = t.employee_id
                                     and s.to_date > current_date
                                     and de.to_date > current_date
                                     and t.to_date > current_date
                                     ORDER BY e.birth_date DESC LIMIT 2"

*** Here is an example of an complex math query --> If 5 new employees were to be hired in Sales Department with each getting a pay of $50000, then how much
would the average pay of all the employees in Sales department change?
    Sample response: "WITH current_sales_data AS (
                        SELECT
                            SUM(s.amount) AS total_current_salary,
                            COUNT(*) AS current_employee_count
                        FROM employees.salary s
                        JOIN employees.department_employee de ON s.employee_id = de.employee_id
                        JOIN employees.department d ON de.department_id = d.id
                        WHERE d.dept_name = 'Sales'
                        AND de.to_date > CURRENT_DATE
                        AND s.to_date > CURRENT_DATE
                    )
                    SELECT
                        ( (total_current_salary + 250000) / (current_employee_count + 5) ) -
                        (total_current_salary / current_employee_count) AS average_pay_change
                    FROM current_sales_data"

*** Here is an example of an complex math query --> Can you find the difference in expenses when it comes to salaries that
are paid to employees that work in Marketing and Sales department?
    Sample response: "WITH dept_salaries AS (
                        SELECT
                            d.dept_name,
                            SUM(s.amount) AS total_salary
                        FROM employees.salary s
                        JOIN employees.department_employee de ON s.employee_id = de.employee_id
                        JOIN employees.department d ON de.department_id = d.id
                        WHERE d.dept_name IN ('Marketing', 'Sales')
                          AND de.to_date > CURRENT_DATE
                          AND s.to_date > CURRENT_DATE
                        GROUP BY d.dept_name
                    )
                    SELECT
                        COALESCE(MAX(CASE WHEN dept_name = 'Sales'     THEN total_salary END), 0) -
                        COALESCE(MAX(CASE WHEN dept_name = 'Marketing' THEN total_salary END), 0) AS salary_difference
                    FROM dept_salaries;"
